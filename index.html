/** * BACKEND CCL - VERSIÓN WORKSPACE PRO * Gestiona la sincronización con Sheets, Calendar y Gmail. */ const SPREADSHEET_ID = '1f9LQ5ck3KnWFqbUbEqK_pRnxn3QEZwU8h-151JoKW30'; const TEMPLATE_DOC_ID = '17Du6lylm_e0q9TjfkK8vQhltbyPGYtBoz_MJwznKT4M'; const FOLDER_ID = '1he3iY4MQuKrTgGh_H4s89y3UzCrsc8H5'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const action = data.action; const payload = data.payload; let result; switch (action) { case 'addCase': result = saveCase(payload); break; case 'addAdvance': result = saveAdvance(payload); break; case 'addMeeting': result = saveMeeting(payload); break; case 'moveCase': result = updateCaseStatus(payload); break; case 'generateActa': result = generateDocFromTemplate(payload); break; default: throw new Error('Acción no permitida'); } return createResponse({ status: 'success', data: result }); } catch (err) { return createResponse({ status: 'error', message: err.toString() }); } } function createResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function getOrCreateSheet(name, headers) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#f3f3f3"); } return sheet; } /** * Registra reunión en Sheets y sincroniza con Google Calendar */ function saveMeeting(p) { // 1. Crear evento en Google Calendar const calendar = CalendarApp.getDefaultCalendar(); const startDateTime = new Date(p.date + 'T' + (p.time || '08:00:00')); const endDateTime = new Date(startDateTime.getTime() + (60 * 60 * 1000)); // Duración 1 hora por defecto const eventOptions = { location: p.location || 'Oficina Principal / Virtual', description: 'Sesión oficial del Comité de Convivencia Laboral. Tipo: ' + p.type, guests: p.emails || '' // Lista de correos separados por coma }; const event = calendar.createEvent(p.title, startDateTime, endDateTime, eventOptions); // 2. Registrar en Google Sheets const headers = ['ID', 'Título', 'Fecha', 'Hora', 'Lugar', 'Tipo', 'Estado', 'CalendarLink']; const sheet = getOrCreateSheet('Reuniones', headers); sheet.appendRow([ event.getId(), p.title, p.date, p.time || '08:00', eventOptions.location, p.type, 'Programada', event.getGuestList().length > 0 ? 'Notificados' : 'Sin invitados' ]); // 3. Enviar notificación por Gmail si hay correos if (p.emails) { sendMeetingNotification(p, eventOptions.location); } return { eventId: event.getId() }; } function sendMeetingNotification(p, location) { const subject = `CITACIÓN: ${p.title} - Comité de Convivencia`; const body = `Se le informa que ha sido citado a una sesión del Comité de Convivencia Laboral.\n\n` + `Detalles:\n` + `Asunto: ${p.title}\n` + `Fecha: ${p.date}\n` + `Hora: ${p.time}\n` + `Lugar: ${location}\n\n` + `Por favor, confirme su asistencia a través de la invitación de Google Calendar que llegará a su correo.`; MailApp.sendEmail(p.emails, subject, body); } function saveCase(p) { const headers = ['ID', 'Fecha', 'Título', 'Categoría', 'Prioridad', 'Reportero', 'Vencimiento', 'Estado', 'Descripción']; const sheet = getOrCreateSheet('Casos', headers); sheet.appendRow([p.id, p.date, p.title, p.category, p.priority, p.reporter, p.dueDate, p.status, p.description]); try { MailApp.sendEmail(Session.getActiveUser().getEmail(), "Nuevo Caso CCL: " + p.id, "Se ha registrado un caso: " + p.title); } catch(e) {} return { id: p.id }; } function saveAdvance(p) { const headers = ['ID Caso', 'Fecha Registro', 'Autor', 'Descripción']; const sheet = getOrCreateSheet('Bitacora', headers); sheet.appendRow([p.caseId, new Date(), p.author, p.text]); return true; } function updateCaseStatus(p) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName('Casos'); if (!sheet) return; const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === p.id) { sheet.getRange(i + 1, 8).setValue(p.status); break; } } } /** * Genera un documento basado en plantilla */ function generateDocFromTemplate(payload) { const templateFile = DriveApp.getFileById(TEMPLATE_DOC_ID); const folder = DriveApp.getFolderById(FOLDER_ID); const fileName = `Acta_${payload.type}_${payload.caseId}_${new Date().getTime()}`; const copy = templateFile.makeCopy(fileName, folder); const doc = DocumentApp.openById(copy.getId()); const body = doc.getBody(); body.replaceText('{{TITULO}}', payload.title || ''); body.replaceText('{{ID_CASO}}', payload.caseId || ''); body.replaceText('{{FECHA}}', new Date().toLocaleDateString()); body.replaceText('{{COMPROMISOS}}', payload.commitments || ''); body.replaceText('{{ASISTENTES}}', payload.attendees || ''); doc.saveAndClose(); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName('Actas') || ss.insertSheet('Actas'); sheet.appendRow([payload.caseId, new Date(), payload.type, doc.getUrl()]); return { docUrl: doc.getUrl() }; }
